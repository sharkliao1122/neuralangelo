conda activate neuralangelo_5080
cd /d C:\research\neuralangelo_test

@echo off
setlocal EnableDelayedExpansion
set "SEQUENCE=20260703_nonsmooth_match"
set "DOWNSAMPLE_RATE=4"
set "VIDEO=C:\research\data_mp4\20260703_nonsmooth_match.MP4"

set "SCENE_TYPE=outdoor"
set "VAL_SHORT_SIZE=300"
set "WORK_ROOT=C:\research\neuralangelo_test"
set "FFMPEG=C:\tools\ffmpeg\bin\ffmpeg.exe"
set "COLMAP=C:\tools\COLMAP\COLMAP.bat"
set "PYTHON=C:\Users\s7103\anaconda3\envs\neuralangelo_5080\python.exe"
set "DATASET_NAME=%SEQUENCE%_ds%DOWNSAMPLE_RATE%"
set "DATA_PATH=%WORK_ROOT%\datasets\%DATASET_NAME%"
set "IMAGE_RAW=%DATA_PATH%\images_raw"
set "CONFIG_NAME=%DATASET_NAME%"
set "CONFIG_PATH=%WORK_ROOT%\projects\neuralangelo\configs\custom\%CONFIG_NAME%.yaml"

if not exist "%DATA_PATH%" mkdir "%DATA_PATH%"
if not exist "%IMAGE_RAW%" mkdir "%IMAGE_RAW%"
if not exist "%DATA_PATH%\sparse" mkdir "%DATA_PATH%\sparse"

rem If you rerun on the same DATA_PATH, remove old sparse\0, sparse\1, ...
rem first to avoid selecting an older larger model by mistake.

"%FFMPEG%" -nostdin -i "%VIDEO%" -vf "select=not(mod(n\,%DOWNSAMPLE_RATE%))" -vsync vfr -q:v 2 "%IMAGE_RAW%\%06d.jpg"

call "%COLMAP%" feature_extractor ^
  --database_path "%DATA_PATH%\database.db" ^
  --image_path "%IMAGE_RAW%" ^
  --ImageReader.camera_model SIMPLE_RADIAL ^
  --ImageReader.single_camera 1 ^
  --FeatureExtraction.use_gpu 1 ^
  --FeatureExtraction.num_threads 32

call "%COLMAP%" sequential_matcher ^
  --database_path "%DATA_PATH%\database.db" ^
  --FeatureMatching.use_gpu 1

call "%COLMAP%" mapper ^
  --database_path "%DATA_PATH%\database.db" ^
  --image_path "%IMAGE_RAW%" ^
  --output_path "%DATA_PATH%\sparse"

set "BEST_SPARSE_DIR="
set /a BEST_SPARSE_SIZE=-1
for /d %%D in ("%DATA_PATH%\sparse\*") do (
  if exist "%%~fD\images.bin" (
    set /a MODEL_SIZE=0
    for %%F in ("%%~fD\*.bin") do set /a MODEL_SIZE+=%%~zF
    if !MODEL_SIZE! GTR !BEST_SPARSE_SIZE! (
      set "BEST_SPARSE_DIR=%%~fD"
      set /a BEST_SPARSE_SIZE=!MODEL_SIZE!
    )
  )
)

if not defined BEST_SPARSE_DIR (
  echo No valid sparse model was found under "%DATA_PATH%\sparse".
  exit /b 1
)

echo Using sparse model "!BEST_SPARSE_DIR!" with total .bin size !BEST_SPARSE_SIZE! bytes.
copy "!BEST_SPARSE_DIR!\*.bin" "%DATA_PATH%\sparse\" /Y

call "%COLMAP%" image_undistorter ^
  --image_path "%IMAGE_RAW%" ^
  --input_path "%DATA_PATH%\sparse" ^
  --output_path "%DATA_PATH%" ^
  --output_type COLMAP

"%PYTHON%" "%WORK_ROOT%\projects\neuralangelo\scripts\convert_data_to_json.py" ^
  --data_dir "%DATA_PATH%" ^
  --scene_type %SCENE_TYPE%

"%PYTHON%" "%WORK_ROOT%\projects\neuralangelo\scripts\generate_config.py" ^
  --sequence_name %CONFIG_NAME% ^
  --data_dir "%DATA_PATH%" ^
  --scene_type %SCENE_TYPE% ^
  --val_short_size %VAL_SHORT_SIZE% ^
  --auto_exposure_wb
